# Any 64-bit Raspberry Pi: 3, 3B+, Zero 2 W, 4, 400, CM4, 5, 500, CM5 (one
# kernel, every DTB; the firmware picks the DTB and start.elf/start4.elf
# by board, and maps overlays to their Pi 5 versions itself). Made for the
# RasPad 3 (a Pi 4B, 1280x800 HDMI panel, USB touch); see
# board/rpi64/README.md.

kernel=Image
arm_64bit=1

# The display through the kernel's KMS driver: octopus-wm drives it with
# wlroots' DRM backend, rendering on the GPU (Mesa; its buffers come from
# the kernel's CMA pool, which vc4-kms-v3d sets up).
dtoverlay=vc4-kms-v3d
max_framebuffers=2
gpu_mem=64
disable_overscan=1

# The 3.5 mm jack where there is one; HDMI audio comes with vc4-kms-v3d.
dtparam=audio=on

# Serial console on the GPIO header: pins 8 (TXD, GPIO14), 10 (RXD,
# GPIO15), 6 (GND); 3.3 V levels, 115200 8N1. The full PL011 UART rather
# than the mini UART: Bluetooth, which would otherwise have it, isn't used.
enable_uart=1
dtoverlay=disable-bt
# The firmware's own boot log on it too (before the kernel).
uart_2ndstage=1

disable_splash=1
boot_delay=0

[pi5]
# The Pi 5's console is its debug connector unless told: the GPIO pins.
dtparam=uart0_console=on

[all]
